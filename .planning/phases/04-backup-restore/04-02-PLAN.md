```typescript
import {spawn} from "node:child_process"

export interface ResticRunOptions {

    s3Endpoint?: string
    s3Bucket?: string

    password: string  // already decrypted
}

export interface RetentionPolicy {

    keepWeekly: number
    keepMonthly: number
}
```

Class `ResticExecutor`:
- Constructor: accepts optional `binary` string, defaults to `process.env.RESTIC_BINARY ?? "restic"`
- `async run(options: ResticRunOptions): Promise<ResticRunResult>` — spawns the binary with `stdio: ["ignore", "pipe", "pipe"]`, env merged with `process.env`. Line-buffers stdout (split on `\n`, flush partial on close). Collects stderr into a string. Resolves with `{exitCode, stderr}`. Rejects on spawn `error` event.
- `buildBackupArgs(stackPath: string, stackId: string): string[]` — returns `["backup", stackPath, "--exclude", `${stackPath}/logs`, "--tag", stackId, "--json"]`
- `buildForgetArgs(stackId: string, policy: RetentionPolicy): string[]` — returns `["forget", "--tag", stackId, "--keep-daily", String(policy.keepDaily), "--keep-weekly", String(policy.keepWeekly), "--keep-monthly", String(policy.keepMonthly), "--prune"]`
- `buildRestoreArgs(snapshotId: string): string[]` — returns `["restore", snapshotId, "--target", "/"]`
- `buildInitArgs(): string[]` — returns `["init"]`
- `async snapshots(repoConfig: BackupRepoConfig, tag: string): Promise<ResticSnapshot[]>` — runs `restic snapshots --tag <tag> --json`, collects all lines, joins them, parses JSON. Returns `[]` if exit code is 10. Throws `Error("restic snapshots failed: <stderr>")` for other non-zero exit codes.
- `buildEnv(config: BackupRepoConfig): Record<string, string>` — builds `RESTIC_REPOSITORY` (via `buildRepoUrl`) and `RESTIC_PASSWORD` from config. Adds `AWS_ACCESS_KEY_ID` and `AWS_SECRET_ACCESS_KEY` for S3 type. For SFTP, adds `RESTIC_REPOSITORY` as `sftp:user@host:path`. For local, `RESTIC_REPOSITORY` is the raw path.
- `buildRepoUrl(config: BackupRepoConfig): string` — local: `config.repoPath`, sftp: `sftp:${config.sftpUser}@${config.sftpHost}:${config.repoPath ?? "/backups"}`, s3: `s3:${config.s3Endpoint ?? "s3.amazonaws.com"}/${config.s3Bucket}`
- `async checkVersion(): Promise<{available: boolean; version?: string}>` — runs `restic version`, parses version string from stdout, returns `{available: true, version}` on success, `{available: false}` on error.

All methods that accept a `BackupRepoConfig` must NEVER put password or keys on the command line -- only in the env object.

Export singleton: `export const resticExecutor = new ResticExecutor()`

Class `BackupRepository`:

- `async create(data: {stackId: string; trigger: BackupTrigger; status: BackupStatus; startedAt: Date; resticSnapshotId?: string})` — creates a Backup record with `prisma.backup.create()`. Set `resticSnapshotId` to `data.resticSnapshotId ?? ""` (empty string, updated after backup completes). Returns the created record.

- `async findByIdOrThrow(id: string)` — uses `prisma.backup.findUnique({where: {id}})`, throws `NotFoundError("Backup \"${id}\" not found")` if null. Returns the record.

- `async findByStackId(stackId: string, limit = 50)` — `prisma.backup.findMany({where: {stackId}, orderBy: {startedAt: "desc"}, take: limit})`. Returns the list.

- `async updateStatus(id: string, data: {status: BackupStatus; completedAt?: Date; errorMessage?: string; sizeBytes?: bigint; resticSnapshotId?: string})` — `prisma.backup.update({where: {id}, data})`.

- `async updateLogLines(id: string, lines: string[])` — `prisma.backup.update({where: {id}, data: {logLines: lines}})`.

- `async findInProgress()` — `prisma.backup.findMany({where: {status: "IN_PROGRESS"}})`. Used for startup recovery.

- `toDto(backup: {sizeBytes: bigint | null; [key: string]: unknown}): Record<string, unknown>` — returns a new object with `sizeBytes` converted from `BigInt` to `String` (or null). Spreads all other fields. This prevents `JSON.stringify` BigInt errors.

Export singleton: `export const backupRepository = new BackupRepository()`
