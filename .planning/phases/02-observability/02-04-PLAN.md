```typescript
import semver from "semver"

export function normalizeImageRef(imageRef: string): string {
    // Strip docker.io/library/ or docker.io/ prefix
    let ref = imageRef
        .replace(/^docker\.io\/library\//, "")
        .replace(/^docker\.io\//, "")
    // Append :latest if no tag present (no colon in final path component)
    if (!ref.includes(":")) ref = ref + ":latest"
    return ref
}

export function detectRegistry(imageRef: string): "dockerhub" | "ghcr" | "private" {
    const normalized = normalizeImageRef(imageRef)
    const host = normalized.split("/")[0]
    // Single component or no dots/colons in host = Docker Hub
    if (!normalized.includes("/") || !host.includes(".")) return "dockerhub"
    if (host === "ghcr.io") return "ghcr"
    return "private"
}

export function parseDateTag(tag: string): Date | null {
    const DATE_PATTERNS = [
        /^(\d{4})-(\d{2})-(\d{2})$/,
        /^(\d{4})(\d{2})(\d{2})$/,
        /^(\d{4})(\d{2})$/,
    ]
    for (const pattern of DATE_PATTERNS) {
        const m = pattern.exec(tag)
        if (m) {
            const d = new Date(`${m[1]}-${m[2] ?? "01"}-${m[3] ?? "01"}`)
            if (!isNaN(d.getTime())) return d
        }
    }
    return null
}

export type CompareResult = "newer" | "same" | "older" | "unknown"

export function compareVersions(currentTag: string, latestTag: string): CompareResult {
    // 1. Try semver (with coerce for truncated tags like "28", "1.25")
    const current = semver.coerce(currentTag)
    const latest = semver.coerce(latestTag)
    if (current && latest) {
        if (semver.gt(latest, current)) return "newer"
        if (semver.eq(latest, current)) return "same"
        return "older"
    }
    // 2. Try date tag
    const currentDate = parseDateTag(currentTag)
    const latestDate = parseDateTag(latestTag)
    if (currentDate && latestDate) {
        if (latestDate > currentDate) return "newer"
        if (latestDate.getTime() === currentDate.getTime()) return "same"
        return "older"
    }
    // 3. Unknown — caller should fall back to digest comparison
    return "unknown"
}
```

**3. UpdateChecker class:**

```typescript
const CHECK_INTERVAL_MS = 6 * 60 * 60 * 1000  // 6 hours

export class UpdateChecker {
    private cronTask: cron.ScheduledTask | null = null
    private readonly repo: UpdateCheckerRepo | null
    private readonly docker: Pick<DockerExecutor, "manifestInspect" | "pull" | "up">
    private readonly broadcaster: Pick<StateBroadcaster, "publish">

    constructor(repo?: UpdateCheckerRepo, docker?: ..., broadcaster?: ...) {
        this.repo = repo ?? null
        this.docker = docker ?? dockerExecutor
        this.broadcaster = broadcaster ?? stateEventBroadcaster
    }

    async start(): Promise<void> {
        // Schedule a cron that fires every 5 minutes and processes one image per tick
        this.cronTask = cron.schedule("*/5 * * * *", async () => {
            try { await this.checkNextImage() }
            catch (err) { console.error("[UpdateChecker] error:", err) }
        })
        console.log("[UpdateChecker] started — checking every 5 minutes, staggered over 6-hour window")
    }

    stop(): void {
        this.cronTask?.stop()
        this.cronTask = null
    }

    async checkNextImage(): Promise<void> {
        const repo = await this.getRepo()
        const imageRefs = await repo.findAllImageRefs()
        if (imageRefs.length === 0) return

        const imageUpdateCheckRepo = await this.getImageUpdateCheckRepo()
        const staggerMs = CHECK_INTERVAL_MS / imageRefs.length
        const cutoff = new Date(Date.now() - staggerMs)

        // Find never-checked images first, then oldest-checked
        const neverChecked = imageRefs.filter(async ref => {
            const record = await imageUpdateCheckRepo.findByImageRef(ref)
            return record === null
        })
        // Use DB query for efficiency
        const due = await imageUpdateCheckRepo.findDueForCheck(cutoff, imageRefs)
        // Pick first due image (null lastCheckedAt sorts first)
        const imageRef = due[0]?.imageRef ?? null
        if (!imageRef) {
            console.log("[UpdateChecker] all images checked within stagger window, skipping")
            return
        }

        await this.checkImage(imageRef, imageUpdateCheckRepo)
    }

    async checkImage(imageRef: string, imageUpdateCheckRepo?: any): Promise<void> {
        const repo = imageUpdateCheckRepo ?? await this.getImageUpdateCheckRepo()
        console.log(`[UpdateChecker] checking ${imageRef}`)

        try {
            const result = await this.docker.manifestInspect(imageRef)
            if (!result) {
                await repo.upsert({
                    imageRef,
                    lastCheckedAt: new Date(),
                    hasUpdate: false,
                    checkError: "manifest inspect returned null",
                })
                return
            }

            const {digest: latestDigest, latestTag} = result
            // Get current digest from Service table
            const serviceRepo = await this.getRepo()
            const stacks = await serviceRepo.findStacksByImageRef(imageRef)

            // For latest tag: digest comparison only
            const tag = imageRef.split(":")[1] ?? "latest"
            let hasUpdate = false

            if (latestTag && tag !== "latest") {
                const comparison = compareVersions(tag, latestTag)
                hasUpdate = comparison === "newer"
            } else if (latestDigest) {
                // Digest comparison for :latest or unknown tags
                const existing = await repo.findByImageRef(imageRef)
                hasUpdate = existing?.latestDigest !== null && existing?.latestDigest !== latestDigest
            }

            await repo.upsert({
                imageRef,
                lastCheckedAt: new Date(),
                latestTag: latestTag ?? null,
                latestDigest: latestDigest ?? null,
                hasUpdate,
                checkError: null,
            })

            // Broadcast update_available for each stack using this image
            if (hasUpdate) {
                for (const stack of stacks) {
                    this.broadcaster.publish({
                        type: "update_available",
                        stackId: stack.id,
                        imageRef,
                        latestTag: latestTag ?? null,
                        hasUpdate: true,
                    })
                }
            }
        } catch (err: any) {
            const checkError = err.message ?? String(err)
            await repo.upsert({
                imageRef,
                lastCheckedAt: new Date(),
                hasUpdate: false,
                checkError,
            })
            console.error(`[UpdateChecker] failed to check ${imageRef}:`, err)
        }
    }
}

export const updateChecker = new UpdateChecker()
```

async manifestInspect(imageRef: string): Promise<{
    digest: string | null
    latestTag: string | null
} | null> {
    try {
        const {stdout} = await execFileAsync(
            "docker",
            ["manifest", "inspect", "--verbose", imageRef],
            {timeout: 30_000},
        )
        // docker manifest inspect --verbose output:
        // Multi-arch: returns JSON array; single-arch: returns JSON object
        const parsed = JSON.parse(stdout)
        const manifest = Array.isArray(parsed) ? parsed[0] : parsed

        // Try to extract digest from known field locations:
        const digest: string | null =
            manifest?.Descriptor?.digest ??           // multi-arch descriptor
            manifest?.SchemaV2Manifest?.config?.digest ??  // single-arch v2
            manifest?.Ref ??
            null

        return {digest, latestTag: null}  // tag listing requires separate registry API
    } catch (err: any) {
        // "no such manifest" = image not found in registry
        if (err.stderr?.includes("no such manifest") || err.stderr?.includes("not found")) {
            return null
        }
        // 429 rate limit or auth error — re-throw so UpdateChecker can record checkError
        throw err
    }
}
