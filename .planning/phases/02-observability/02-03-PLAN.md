export class StatePoller {
    private cronTask: cron.ScheduledTask | null = null
    private readonly broadcaster: Pick<StateBroadcaster, "publish">

    constructor(
        docker?: ...,
        repo?: StatePollerRepo,
        broadcaster?: Pick<StateBroadcaster, "publish">,
    ) {
        // repo stored as-is; lazy getRepo() avoids db.ts in module graph
        this.repo = repo ?? null
        this.broadcaster = broadcaster ?? stateEventBroadcaster
    }

    private async getRepo(): Promise<StatePollerRepo> {
        if (this.repo !== null) return this.repo
        const {stackRepository} = await import("../repositories/stack-repository.js")
        return stackRepository as unknown as StatePollerRepo
    }

    async start(): Promise<void> {
        await this.startEventStream()
        this.cronTask = cron.schedule("*/60 * * * * *", async () => {
            try { await this.reconcile() } catch (err) { console.error(err) }
        })
    }

    stop(): void {
        this.cronTask?.stop()
        this.cronTask = null
    }
}

**FileWatcherRepo interface** (constructor-injectable, matches what test scaffold expects):
```typescript
export interface FileWatcherRepo {
    findAllStacks(): Promise<Array<{id: string; hostPath: string; lastKnownHash: string | null}>>
    findStackByPath(composePath: string): Promise<{id: string; lastKnownHash: string | null} | null>
    updateStackHash(stackId: string, hash: string): Promise<void>
    clearConfigChanged(stackId: string): Promise<void>
    setConfigChanged(stackId: string, changed: boolean): Promise<void>
}
```

**Implementation details:**
- `STACKS_ROOT = process.env.STACKS_ROOT ?? "/stacks"` (consistent with rest of codebase)
- `findStackByPath` matches on: `stack.hostPath + "/docker-compose.yml"` equals the watched filePath
- `handleFileChange(filePath)`: use `fs.readFile(filePath, "utf-8")`, then `hashComposeContent(content)`, compare to `stack.lastKnownHash`
  - If hash equal: log and return (no-op)
  - Try `parseComposeContent(content)` — if this throws (invalid YAML or no services key) → createEvent config_error with error.message, broadcast ConfigErrorEvent
  - If valid: `updateStackHash(stack.id, newHash)`, `setConfigChanged(stack.id, true)`, then call `stackEventRepository.createEvent({stackId, type: "config_changed", payload: JSON.stringify({oldHash, newHash})})`, then `broadcaster.publish({type:"config_changed", stackId, newHash})`
- `reconcile()`: call `findAllStacks()`, for each stack: `const filePath = stack.hostPath + "/docker-compose.yml"`, try `readFile`, on ENOENT skip; call `handleFileChange(filePath)` only if hash differs (to avoid duplicate events)
- chokidar `watcher.on("change", (filePath) => this.handleFileChange(filePath).catch(err => console.error(...)))`
- chokidar `watcher.on("add", ...)` same handler (handles externally added stacks)
- chokidar `watcher.on("error", err => console.error("[FileWatcher] chokidar error:", err))`
- Use lazy dynamic import for both repo and stackEventRepository:
```typescript

    if (this.repo !== null) return this.repo
    const {stackRepository} = await import("../repositories/stack-repository.js")
    return stackRepository as unknown as FileWatcherRepo
}
private async getEventRepo() {
    const {stackEventRepository} = await import("../repositories/stack-event-repository.js")
    return stackEventRepository
}
```
